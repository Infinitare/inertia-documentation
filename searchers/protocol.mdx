## searcher.proto

```protobuf
syntax = "proto3";

package inertia.searcher.v1;

// Every call authenticates with the API key in the `x-api-key` metadata.
service Searcher {
  // Streams the transactions and bundles that are up for auction on their way to the validators, batched. Requires
  // orderflow access.
  rpc SubscribeOrderflow(SubscribeOrderflowRequest) returns (stream Bundles);

  // Bids on streamed bundles and streams back what happened to every bid. Requires intake access.
  rpc SubmitBundles(stream Bundles) returns (stream BundleUpdate);

  // Streams the transactions the validators behind this node executed in their own leader slots, as they execute and
  // before the block is complete, batched. Requires preconfs access.
  rpc SubscribePreconfs(SubscribePreconfsRequest) returns (stream Preconfs);

  // Returns the upcoming leader windows of the validators behind this node, as far as the leader schedule is known.
  rpc GetLeaderWindows(GetLeaderWindowsRequest) returns (LeaderWindows);
}

message GetLeaderWindowsRequest {}

message LeaderWindow {
  uint64 first_slot = 1;
  uint64 last_slot = 2;
}

message LeaderWindows {
  // Slot the node was at when answering.
  uint64 slot = 1;
  // At most 10, in slot order. The window of the current slot is included while it lasts. Fewer are returned when the
  // leader schedule does not reach further.
  repeated LeaderWindow windows = 2;
}

message SubscribeOrderflowRequest {
  // 1 to 1024 32-byte pubkeys, static or loaded from a lookup table. A bundle matches if any of its transactions
  // touches one of them.
  repeated bytes accounts = 1;
}

message SubscribePreconfsRequest {
  // 1 to 1024 32-byte pubkeys, static or loaded from a lookup table. A transaction matches if it touches one of them.
  repeated bytes accounts = 1;
  // Also streams the transactions that failed. They stay in the block and pay their fee.
  bool include_failed = 2;
}

// A transaction the leader executed and keeps in its block of this slot. A failed one has success = false and is only
// streamed with include_failed.
message Preconf {
  uint64 slot = 1;
  bytes signature = 2;
  bytes transaction = 3;
  bool success = 4;
}

message Preconfs {
  repeated Preconf preconfs = 1;
}

// Streamed: the signature of the first transaction and all transactions of the bundle, one for a plain transaction.
// Submitted: the signature of the streamed bundle bid on and only the searcher's own transactions, which have to
// transfer the tip to one of the tip accounts. Both together may not exceed 5 transactions.
message Bundle {
  bytes signature = 1;
  repeated bytes transactions = 2;
}

message Bundles {
  repeated Bundle bundles = 1;
}

message BundleUpdate {
  // Signature of the streamed bundle the bid is on.
  bytes signature = 1;
  // Signature of the first transaction of the bid.
  bytes bid_signature = 2;
  oneof update {
    Outbid outbid = 3;
    Rejected rejected = 4;
    Sent sent = 5;
    Landed landed = 6;
    Dropped dropped = 7;
  }
}

// Another bid tips more, or the bid did not tip more than the winning one when it arrived. The bid can still be sent
// as a backup, see Sent.
message Outbid {
  uint64 tip = 1;
}

enum Reason {
  REASON_UNSPECIFIED = 0;
  // The signature is not held for bidding, it is unknown or its auction is closed.
  NOT_HELD = 1;
  // No transactions, or more than 5 together with the streamed bundle.
  BUNDLE_SIZE = 2;
  INVALID = 3;
  UNVERIFIED = 4;
  // A transaction was already seen on its way to the validators. Bids carry the searcher's own transactions only.
  DUPLICATE = 5;
  // No transfer to a tip account.
  NO_TIP = 6;
  // A transaction failed a validator rule, named in the detail.
  REJECTED = 7;
  // The bid queue was full.
  DROPPED = 8;
  // The bid won but no validator took the bundle.
  UNDELIVERABLE = 9;
}

message Rejected {
  Reason reason = 1;
  string detail = 2;
}

// The bid's bundle was sent to the validator. Rank 0 is the winning bid, the next highest bids of other searchers follow
// as ranks 1 and 2 and only land if every bundle before them failed. Only the highest bid of each searcher is kept.
message Sent {
  uint64 tip = 1;
  uint32 rank = 2;
}

// The bid's bundle executed and is in the block of this slot. Follows Sent, as does Dropped; a bid the validator never
// executed, because its bundle expired or was displaced, gets neither.
message Landed {
  uint64 slot = 1;
}

// The validator executed the bid's bundle in this slot and left it out of its block. `failed` is the transaction that
// failed, which may be the streamed one; `error` is the validator's transaction error for it, named in `detail` together
// with the instruction index and the program's custom error code when an instruction failed. Every other transaction of
// the bundle failed as its peer.
message Dropped {
  uint64 slot = 1;
  bytes failed = 2;
  int32 error = 3;
  int32 instruction_error = 4;
  uint32 instruction = 5;
  uint32 custom = 6;
  string detail = 7;
}
```

## Transaction format

Every entry in `transactions` and every `transaction` of a `Preconf` is one serialized Solana `VersionedTransaction`. These are the bytes you base64-encode for the `sendTransaction` RPC call. The message can be legacy, v0 or v1.

In Rust, `bincode` parses the bytes into `solana_transaction::versioned::VersionedTransaction`, and serializes your bid transactions the same way:

```rust
use solana_transaction::versioned::VersionedTransaction;

let transaction: VersionedTransaction = bincode::deserialize(&bytes)?;
let bytes = bincode::serialize(&transaction)?;
```

This works with `bincode` 1 and `solana-transaction` 4.3 with the `serde` feature, which parses all three message formats.
